import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.cluster import KMeans
------------------------------------------------------
df=pd.read_csv("final_fifa.csv")
df.head()
---------------------------------------------------------
print(df.isnull().sum())
-----------------------------------------------------
df=df.dropna()
-----------------------------------------------------
df.to_csv("abc.csv",index=False)
print("data save")

-----------------------------------------------------
x = df[["age","jersey_number","height_cm","weight_kg"]]
x_array = x.to_numpy()
print(x_array)

----------------------------------------------------------
inertia = []

for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(x_array)
    inertia.append(model.inertia_)

plt.figure(figsize=(8, 5))
plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("Inertia")
plt.title("Elbow Method")
plt.show()
-------------------------------------------------------------------------
kmeans = KMeans(
    n_clusters=3,
    init="k-means++",
    random_state=42,
    n_init=10
)
df["Cluster"] = kmeans.fit_predict(x_array)

print(df.head())

---------------------------------------------------------------------------

plt.figure(figsize=(8,5))

plt.scatter(
    df["height_cm"],
    df["weight_kg"],
    c=df["Cluster"],   
    cmap="viridis"
)

plt.xlabel("Height (cm)")
plt.ylabel("Weight (kg)")
plt.title("K-Means Clustering")
plt.show()
