        {
            double value;
            double result;

            if (!double.TryParse(txtInput.Text, out value))
            {
                MessageBox.Show("Please enter a vlid numeri value.");
                return;
            }

            if (cmbConversion.SelectedIndex == -1)
            {
                MessageBox.Show("Please select a conversion type.");
                return;
            }

            String conversion = cmbConversion.SelectedItem.ToString();

            switch (conversion)
            {
                case "Centimeters to Inches":
                    result = value / 2.54;
                    lblResult.Text = "Result:" + result.ToString("0.00" + "inches");
                    break;

                case "Inches to Centimeters":
                    result = value * 2.54;
                    lblResult.Text = "Result: " + result.ToString("0.00") + " cm";
                    break;

                case "Kilograms to Pounds":
                    result = value * 2.20462;
                    lblResult.Text = "Result: " + result.ToString("0.00") + " pounds";
                    break;


                case "Pounds to Kilograms":
                    result = value / 2.20462;
                    lblResult.Text = "Result: " + result.ToString("0.00") + " kg";
                    break;

                case "Celsius to Fahrenheit":
                    result = (value * 9 / 5) + 32;
                    lblResult.Text = "Result: " + result.ToString("0.00") + " °F";
                    break;

                case "Fahrenheit to Celsius":
                    result = (value - 32) * 5 / 9;
                    lblResult.Text = "Result: " + result.ToString("0.00") + " °C";
                    break;



            }

        }

        private void btnClear_Click(object sender, EventArgs e)
        {
            txtInput.Clear();
            cmbConversion.SelectedIndex = -1;

            public DateTime TimeAdded { get; set; }

            internal object OrdeBy(Func<object, int> value)
            {

            }
        }

        List<Patient> patients = new List<Patient>();
        public Form1()
        {
            InitializeComponent();
            this.Load += Form1_Load;
        }

        private void Form1_Load(object sender, EventArgs e)
        {
            cmbPriority.Items.Add("Emergency");
            cmbPriority.Items.Add("Urgent");
            cmbPriority.Items.Add("Nomal");

            lblNowServing.Text = "Now Serving None";
        }

        private void btnAddQueue_Click(object sender, EventArgs e)
        {
            if(string.IsNullOrWhiteSpace(txtPatientName.Text) ||
                cmbPriority.SelectedIndex == -1)

            {
                MessageBox.Show("Please enter patient name and select priority.");
                return;

            }

            Patient patient = new Patient
            {
                Name = txtPatientName.Text,
                Priority = cmbPriority.SelectedItem.ToString(),
                TimeAdded = DateTime.Now

            };

            patients.Add(patient);


            patients = patients
                .OrderBy(p => GetPriorityValue(p.Priority))
                .ThenBy(p => p.TimeAdded)
                .ToList();

            RefreshQueue();

            txtPatientName.Clear();
            cmbPriority.SelectedIndex = -1;


        }

        private int GetPriorityValue(object priority)
        {
            throw new NotImplementedException();
        }

        private void btnCallNext_Click(object sender, EventArgs e)
        {
            if (patients.Count == 0)
            {
                MessageBox.Show("No patients are currently in the queue.");
                return;
            }

            Patient nextPatient = patients[0];

            MessageBox.Show(
                $"Calling: {nextPatient.Name}\nPriority: {nextPatient.Priority}"
            );

            patients.RemoveAt(0);

            RefreshQueue();
        }

        // Priority Order
        private int GetPriorityValue(string priority)
        {
            if (priority == "Emergency")
                return 1;
            else if (priority == "Urgent")
                return 2;
            else
                return 3;
        }

        // Refresh Queue Display
        private void RefreshQueue()
        {
            lstQueue.Items.Clear();

            foreach (Patient patient in patients)
            {
                lstQueue.Items.Add(
                    $"{patient.Name} - {patient.Priority} ({patient.TimeAdded:HH:mm:ss})"
                );
            }

            if (patients.Count > 0)
            {
                lblNowServing.Text = "Now Serving: " + patients[0].Name;
            }
            else
            {
                lblNowServing.Text = "Now Serving: None";
            }
        }

       

        private void btnClear_Click(object sender, EventArgs e)
        {
            txtPatientName.Clear();
            cmbPriority.SelectedIndex = -1;

        }
    }

using System;
using System.Windows.Forms;

namespace MedicineDosageCalculator
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();

            // ComboBox values
            cmbDosesPerDay.Items.Add("1");
            cmbDosesPerDay.Items.Add("2");
            cmbDosesPerDay.Items.Add("3");
            cmbDosesPerDay.Items.Add("4");

            // Initial values
            numWeight.Value = 0;
            numDosage.Value = 0;

            lblDoseResult.Text = "0 mg";
            lblDailyResult.Text = "0 mg";
            lblWarning.Text = "";
        }

        // Calculate Dosage Button
        private void btnCalculate_Click(object sender, EventArgs e)
        {
            // Validate Patient Name
            if (string.IsNullOrWhiteSpace(txtPatientName.Text))
            {
                MessageBox.Show(
                    "Please enter patient's name.",
                    "Input Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning
                );

                txtPatientName.Focus();
                return;
            }

            // Validate Weight
            if (numWeight.Value <= 0)
            {
                MessageBox.Show(
                    "Weight must be greater than zero.",
                    "Input Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning
                );

                numWeight.Focus();
                return;
            }

            // Validate Dosage
            if (numDosage.Value <= 0)
            {
                MessageBox.Show(
                    "Dosage must be greater than zero.",
                    "Input Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning
                );

                numDosage.Focus();
                return;
            }

            // Validate Doses Per Day
            if (cmbDosesPerDay.SelectedIndex == -1)
            {
                MessageBox.Show(
                    "Please select doses per day.",
                    "Input Error",
                    MessageBoxButtons.OK,
                    MessageBoxIcon.Warning
                );

                cmbDosesPerDay.Focus();
                return;
            }

            // Get input values
            decimal weight = numWeight.Value;
            decimal dosage = numDosage.Value;

            int dosesPerDay = int.Parse(
                cmbDosesPerDay.SelectedItem.ToString()
            );

            // Calculate dose per administration
            decimal dosePerAdministration = weight * dosage;

            // Calculate total daily dosage
            decimal totalDailyDosage =
                dosePerAdministration * dosesPerDay;

            // Display results
            lblDoseResult.Text =
                dosePerAdministration.ToString("0.##") + " mg";

            lblDailyResult.Text =
                totalDailyDosage.ToString("0.##") + " mg";

            // Safety warning
            if (totalDailyDosage > 1000)
            {
                lblWarning.Text =
                    "Warning: Please consult the doctor.";
            }
            else
            {
                lblWarning.Text = "";
            }
        }

        // Clear Button
        private void btnClear_Click(object sender, EventArgs e)
        {
            // Clear patient name
            txtPatientName.Clear();

            // Reset NumericUpDown controls
            numWeight.Value = 0;
            numDosage.Value = 0;

            // Reset ComboBox
            cmbDosesPerDay.SelectedIndex = -1;

            // Reset results
            lblDoseResult.Text = "0 mg";
            lblDailyResult.Text = "0 mg";

            // Clear warning
            lblWarning.Text = "";

            // Put cursor back into Patient Name
            txtPatientName.Focus();
        }
    }
}

using System;
using System.Windows.Forms;

namespace PatientBMICalculator
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void btnCalculate_Click(object sender, EventArgs e)
        {
            if (string.IsNullOrWhiteSpace(txtName.Text) ||
                string.IsNullOrWhiteSpace(txtRegNo.Text) ||
                string.IsNullOrWhiteSpace(txtAge.Text) ||
                string.IsNullOrWhiteSpace(txtHeight.Text) ||
                string.IsNullOrWhiteSpace(txtWeight.Text))
            {
                MessageBox.Show("Please fill all fields.");
                return;
            }

            double age;
            double height;
            double weight;

            if (!double.TryParse(txtAge.Text, out age) ||
                !double.TryParse(txtHeight.Text, out height) ||
                !double.TryParse(txtWeight.Text, out weight))
            {
                MessageBox.Show("Please enter valid numeric values.");
                return;
            }

            if (age <= 0 || height <= 0 || weight <= 0)
            {
                MessageBox.Show("Age, Height and Weight must be greater than zero.");
                return;
            }

            double bmi = weight / (height * height);

            string category;

            if (bmi < 18.5)
            {
                category = "Underweight";
            }
            else if (bmi < 25)
            {
                category = "Normal";
            }
            else if (bmi < 30)
            {
                category = "Overweight";
            }
            else
            {
                category = "Obese";
            }

            lblBMI.Text = "BMI: " + bmi.ToString("0.00");
            lblCategory.Text = "Health Category: " + category;

            lstPatients.Items.Add(
                txtRegNo.Text + " - " +
                txtName.Text + " - BMI: " +
                bmi.ToString("0.00") + " - " +
                category
            );
        }

        private void btnClear_Click(object sender, EventArgs e)
        {
            txtName.Clear();
            txtRegNo.Clear();
            txtAge.Clear();
            txtHeight.Clear();
            txtWeight.Clear();

            lblBMI.Text = "BMI:";
            lblCategory.Text = "Health Category:";
        }
    }
}
